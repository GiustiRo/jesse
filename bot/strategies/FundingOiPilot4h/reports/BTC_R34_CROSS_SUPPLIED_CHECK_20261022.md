def supplied(name, names=None):
    tree = ast.parse(SOURCES[name], filename="<supplied:"+name+">")
    nodes = [node for node in tree.body if isinstance(node,(ast.FunctionDef,ast.ClassDef))
             and (names is None or node.name in names)]
    if names is not None:
        assert len(nodes) == len(names)
    exec(compile(ast.Module(body=nodes,type_ignores=[]),
                 "<supplied:"+name+">","exec"),globals())

def rows_for_cross(side=1, tiny=False, first=0):
    # Same hand-computable mean fixture as archived CROSS:60 higher closes,
    #20 lower closes and a fresh resumption. Tiny version exercises 1x sizing.
    values = ([100.]*60 + [99.]*20 + [130.])
    if tiny:
        values = [100.+(x-100.)*.001 for x in values]
    values = np.array(values)
    if side == -1:
        values = 400.-values
    spread = .01 if not tiny else .00001
    return np.column_stack((first+np.arange(81)*BAR,values,values,
                            values+spread,values-spread,np.ones(81)))
