```python
import json,collections
from pathlib import Path
m=json.load(open('inputs/manifest.json'))
d={x['name']:json.load(open(x['file'])) for x in m}
def index(data):
 result={};ambiguous=0
 for t in data.get('trades',[]):
  orders=[o for o in t['orders'] if o['status']=='EXECUTED' and not o.get('reduce_only')]
  if len(orders)!=1 or orders[0].get('created_at') is None:
   ambiguous+=1;continue
  key=(t['symbol'],t['type'],orders[0]['created_at'])
  result.setdefault(key,[]).append((t,orders[0]))
 valid={k:v[0] for k,v in result.items() if len(v)==1}
 ambiguous+=sum(len(v) for v in result.values() if len(v)!=1)
 return valid,ambiguous
a,ba=index(d['Cq']);b,bb=index(d['M'])
rows=[]
for k in sorted(set(a)&set(b)):
 x,ox=a[k];y,oy=b[k]
 nx=abs(x['qty'])*x['entry_price'];ny=abs(y['qty'])*y['entry_price']
 rows.append({'created_at':k[2],'symbol':k[0],'side':k[1],'opened_Cq':x['opened_at'],'opened_M':y['opened_at'],'entry_Cq':x['entry_price'],'entry_M':y['entry_price'],'qty_Cq':x['qty'],'qty_M':y['qty'],'PNL_Cq':x['PNL'],'PNL_M':y['PNL'],'PNL_per_entrynotional_Cq':x['PNL']/nx,'PNL_per_entrynotional_M':y['PNL']/ny,'entrytype_Cq':ox['type'],'entrytype_M':oy['type']})
out={'secondary_diagnostic':True,'common_created_entries':len(rows),'validCq':len(a),'validM':len(b),'ambiguousCq':ba,'ambiguousM':bb,'sumPNL_Cq':sum(r['PNL_Cq'] for r in rows),'sumPNL_M':sum(r['PNL_M'] for r in rows),'sum_normalized_Cq':sum(r['PNL_per_entrynotional_Cq'] for r in rows),'sum_normalized_M':sum(r['PNL_per_entrynotional_M'] for r in rows),'different_fill_times':sum(r['opened_Cq']!=r['opened_M'] for r in rows),'rows':rows,'limitation':'Secondary after inspection, same order creation decision conditional on both fills and endogenous occupation. Not opportunities population; primary exact fill pairing retained.'}
Path('output').mkdir(exist_ok=True)
json.dump(out,open('output/decision_pairs.json','w'),indent=2)
print(json.dumps({k:v for k,v in out.items() if k!='rows'}))
```
